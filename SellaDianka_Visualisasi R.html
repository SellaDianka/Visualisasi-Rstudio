4. Berikan label dengan nilai numerik pada setiap chart, dengan mengurutkan dari presentase terbesar ke terkecil, pada sumbu X buat label miring dengan 45 derajat

library(scales)
## 
## Attaching package: 'scales'

##     col_factor
plotdata=heart%>%
  count(ChestPain)%>%
  mutate(pct=n/sum(n),
         pctlabel=paste0(round(pct*100), '%'))
ggplot(plotdata, aes(x=reorder(ChestPain, -pct),
                     y=pct))+
  geom_bar(stat="Identity", fill=rainbow(4), color="azure4")+
  geom_text(aes(label=pctlabel), 
            vjust=0.25)+
  scale_y_continuous(labels=scales::percent)+
  theme_classic()+
  labs(x="Nyeri Dada", y="Frekuensi", title="Heart")+
  theme(axis.text.x = element_text(angle = 45, hjust = 1.25))

Kode tersebut digunakan untuk membuat grafik batang yang menunjukkan frekuensi jenis nyeri dada dalam dataset “heart” dengan pengaturan tampilan grafik, label sumbu, dan label persentase menggunakan fungsi ‘scales’, serta memiringkan label sumbu x sebesar 45 derajat.
